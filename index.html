구글 시트 & 드라이브 연동 가이드

나만의 구글 시트 및 구글 드라이브 이미지 저장소를 연동하려면 다음 절차를 진행하세요:

  1. 구글 드라이브에서 새 구글 스프레드시트를 생성합니다.
  2. 상단 메뉴 [확장 프로그램] > [Apps Script]를 클릭합니다.
  3. 기존 코드를 모두 지우고 아래 스크립트 코드를 붙여넣어 저장한 뒤 [배포] > [새 배포]를 선택합니다.
  4. 유형: 웹 앱 / 액세스 권한: 모든 사용자(Anyone)로 설정 후 배포합니다.
  5. 발급된 웹 앱 URL을 복사하여 아래 URL 입력 칸에 저장합니다.
function doGet(e) {
  var sp = PropertiesService.getScriptProperties();
  var data = sp.getProperty('TRIP_DATA');
  return ContentService.createTextOutput(data || '{}')
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var contents = JSON.parse(e.postData.contents);
    
    // 1. 이미지 구글 드라이브 업로드 액션
    if (contents.action === 'uploadImage') {
      var folderName = "TripManager_Uploads";
      var folders = DriveApp.getFoldersByName(folderName);
      var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);
      
      var base64Data = contents.base64Data.replace(/^data:image\/\w+;base64,/, "");
      var decoded = Utilities.base64Decode(base64Data);
      var blob = Utilities.newBlob(decoded, contents.mimeType || 'image/jpeg', contents.fileName || 'trip_photo.jpg');
      
      var file = folder.createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      
      var directUrl = "https://lh3.googleusercontent.com/d/" + file.getId();
      return ContentService.createTextOutput(JSON.stringify({ status: 'ok', url: directUrl, fileId: file.getId() }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    // 2. 전체 데이터 시트 저장 액션
    PropertiesService.getScriptProperties().setProperty('TRIP_DATA', JSON.stringify(contents));
    return ContentService.createTextOutput(JSON.stringify({ status: 'ok' }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
